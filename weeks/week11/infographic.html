// Programmatic Python agent run
from antigravity.sdk import Runtime
from cognitive_core.agent import scholar_prime

runtime = Runtime()
session = runtime.create_session(agent=scholar_prime)

response = session.send_message(
"Search arXiv for JAX simulations"
)
print(response.content)
